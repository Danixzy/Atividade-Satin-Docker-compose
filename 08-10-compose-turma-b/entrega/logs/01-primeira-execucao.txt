$ docker compose exec portal cat /etc/nginx/conf.d/default.conf
server {
    listen 80;
    resolver 127.0.0.11 valid=5s;

    location / {
        root /usr/share/nginx/html;
        index index.html;
    }

    location /api/ {
        set $destino http://127.0.0.1:3000;
        rewrite ^/api/(.*)$ /$1 break;
        proxy_pass $destino;
    }
}
